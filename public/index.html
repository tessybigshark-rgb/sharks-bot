import { Router, type Request, type Response } from 'express'; import { generateAIInsight } from './services/aiService'; import { getDashboardSummary } from './services/analyticsService'; import { generateMusicBrief } from './services/musicService'; import { generateSocialCampaign } from './services/socialMediaService'; import { getRevenueIdeas } from './services/revenueService'; import { searchSpotifyTracks } from './services/spotifyService'; import { handleIncomingWhatsAppMessage } from './services/whatsappService'; const router = Router(); router.get('/health', (_req: Request, res: Response) => { res.json({ ok: true, status: 'healthy', app: 'Sharks Bot' }); }); router.get('/dashboard/summary', (_req: Request, res: Response) => { res.json({ success: true, data: getDashboardSummary() }); }); router.post('/music/generate', (req: Request, res: Response) => { const result = generateMusicBrief(req.body ?? {}); res.json({ success: true, data: result }); }); router.post('/social/generate', (req: Request, res: Response) => { const result = generateSocialCampaign(req.body ?? {}); res.json({ success: true, data: result }); }); router.post('/spotify/search', async (req: Request, res: Response) => { const payload = req.body ?? {}; const tracks = await searchSpotifyTracks( Array.isArray(payload.seedGenres) ? payload.seedGenres : ['afrobeats', 'dancehall'], Number(payload.limit ?? 5), ); res.json({ success: true, data: tracks }); }); router.post('/whatsapp/webhook', (req: Request, res: Response) => { const payload = req.body ?? {}; const result = handleIncomingWhatsAppMessage({ message: payload.message ?? '', senderName: payload.senderName ?? 'Fan', }); res.json({ success: true, data: result }); }); router.get('/revenue/ideas', (_req: Request, res: Response) => { res.json({ success: true, data: getRevenueIdeas() }); }); router.post('/ai/insight', async (req: Request, res: Response) => { const prompt = typeof req.body?.prompt === 'string' ? req.body.prompt : 'Create a growth strategy for my music brand.'; const result = await generateAIInsight(prompt); res.json({ success: true, data: result }); }); export default router;
