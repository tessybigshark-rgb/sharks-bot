import cors from 'cors'; import express, { type Request, type Response } from 'express'; import path from 'node:path'; import { envConfig } from './config/env'; import { generateMusicBrief } from './services/musicService'; import { generateSocialCampaign } from './services/socialMediaService'; import { getRevenueIdeas } from './services/revenueService'; import { searchSpotifyTracks } from './services/spotifyService'; import { handleIncomingWhatsAppMessage } from './services/whatsappService'; const app = express(); const publicDir = path.join(__dirname, '..', 'public'); app.use(cors()); app.use(express.json({ limit: '1mb' })); app.use(express.static(publicDir)); app.get('/health', (_req: Request, res: Response) => { res.json({ ok: true, app: envConfig.APP_NAME, status: 'healthy', timestamp: new Date().toISOString(), }); }); app.get('/api/dashboard/summary', (_req: Request, res: Response) => { res.json({ app: envConfig.APP_NAME, stats: { musicCampaigns: 18, activePlaylists: 12, socialPosts: 63, revenueStreams: 5, }, status: 'ready', }); }); app.get('/', (_req: Request, res: Response) => { res.sendFile(path.join(publicDir, 'index.html')); }); app.post('/api/music/generate', (req: Request, res: Response) => { const payload = req.body ?? {}; const result = generateMusicBrief(payload); res.json({ success: true, data: result }); }); app.post('/api/social/generate', (req: Request, res: Response) => { const payload = req.body ?? {}; const result = generateSocialCampaign(payload); res.json({ success: true, data: result }); }); app.post('/api/spotify/search', async (req: Request, res: Response) => { const payload = req.body ?? {}; const seedGenres = Array.isArray(payload.seedGenres) ? payload.seedGenres : ['afrobeats', 'dancehall']; const limit = Number(payload.limit ?? 5); const tracks = await searchSpotifyTracks(seedGenres, limit); res.json({ success: true, data: tracks }); }); app.post('/api/whatsapp/webhook', (req: Request, res: Response) => { const payload = req.body ?? {}; const result = handleIncomingWhatsAppMessage({ message: payload.message ?? '', senderName: payload.senderName ?? 'Fan', }); res.json({ success: true, data: result }); }); app.get('/api/revenue/ideas', (_req: Request, res: Response) => { res.json({ success: true, data: getRevenueIdeas() }); }); app.listen(envConfig.PORT, () => { console.log(`${envConfig.APP_NAME} running on http://localhost:${envConfig.PORT}`); });
